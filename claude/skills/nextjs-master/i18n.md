# Next.js Internationalization (i18n) — Locale Routing, Dictionaries, and Accept-Language

## Overview

Internationalization in Next.js involves URL-based locale routing and localized content. You handle routing via Proxy (detecting the user's preferred locale) and serve localized content via a dictionary system. The key is separating URL routing from content localization.

**Key patterns:**
- **Locale routing** — URLs include locale prefix: `/en/products`, `/fr/products`
- **Accept-Language detection** — Proxy detects preferred locale from headers
- **Sub-path routing** — locale in URL path: `/en/`, `/fr/`, `/de/`
- **Domain routing** — locale via domain: `myapp.fr`, `myapp.de`
- **Dictionaries** — JSON translation files per locale
- **`generateStaticParams`** — prerender all locale variants

**Related skills:** proxy, routing, metadata

## Locale Routing — Sub-Path Pattern

### Proxy Locale Detection

```ts
// proxy.ts
import { NextRequest, NextResponse } from 'next/server'
import { match } from '@formatjs/intl-localematcher'
import Negotiator from 'negotiator'

const locales = ['en-US', 'fr-FR', 'de-DE', 'ja-JP'] as const
const defaultLocale = 'en-US'
type Locale = typeof locales[number]

function getLocale(request: NextRequest): Locale {
  const headers = {
    'accept-language': request.headers.get('accept-language') ?? '',
  }
  const languages = new Negotiator({ headers }).languages()
  return match(languages, [...locales], defaultLocale) as Locale
}

export async function proxy(request: NextRequest) {
  const { pathname } = request.nextUrl

  // Check if pathname already has a locale
  const pathnameHasLocale = locales.some(
    (locale) => pathname.startsWith(`/${locale}/`) || pathname === `/${locale}`
  )

  if (pathnameHasLocale) return NextResponse.next()

  // Redirect to locale-prefixed path
  const locale = getLocale(request)
  const url = request.nextUrl
  url.pathname = `/${locale}${pathname}`
  return NextResponse.redirect(url)
}

export const config = {
  matcher: ['/((?!api|_next/static|_next/image|favicon.ico).*)'],
}
```

### Dictionary Loader

```ts
// app/[lang]/dictionaries.ts
import 'server-only'

const dictionaries: Record<string, () => Promise<Record<string, any>>> = {
  'en-US': () => import('./dictionaries/en-US.json').then(m => m.default),
  'fr-FR': () => import('./dictionaries/fr-FR.json').then(m => m.default),
  'de-DE': () => import('./dictionaries/de-DE.json').then(m => m.default),
  'ja-JP': () => import('./dictionaries/ja-JP.json').then(m => m.default),
}

export const SUPPORTED_LOCALES = Object.keys(dictionaries) as string[]

export function hasLocale(locale: string): locale is keyof typeof dictionaries {
  return locale in dictionaries
}

export const getDictionary = async (locale: string) => {
  if (!hasLocale(locale)) return dictionaries['en-US']()
  return dictionaries[locale]()
}
```

## Localized Layout and Pages

```tsx
// app/[lang]/layout.tsx
import { notFound } from 'next/navigation'
import { hasLocale, getDictionary } from './dictionaries'

type Props = {
  children: React.ReactNode
  params: Promise<{ lang: string }>
}

export async function generateStaticParams() {
  return [
    { lang: 'en-US' },
    { lang: 'fr-FR' },
    { lang: 'de-DE' },
    { lang: 'ja-JP' },
  ]
}

export default async function LocaleLayout({ children, params }: Props) {
  const { lang } = await params

  if (!hasLocale(lang)) notFound()

  const dict = await getDictionary(lang)

  return (
    <html lang={lang}>
      <body>
        <nav>
          <a href={`/${lang}`}>{dict.navigation.home}</a>
          <a href={`/${lang}/products`}>{dict.navigation.products}</a>
        </nav>
        {children}
      </body>
    </html>
  )
}
```

```tsx
// app/[lang]/page.tsx
import { getDictionary } from './dictionaries'

type Props = {
  params: Promise<{ lang: string }>
}

export default async function Page({ params }: Props) {
  const { lang } = await params
  const dict = await getDictionary(lang)

  return (
    <section>
      <h1>{dict.hero.title}</h1>
      <p>{dict.hero.subtitle}</p>
      <a href={`/${lang}/products`}>{dict.cta.shopNow}</a>
    </section>
  )
}
```

## Static Generation for All Locales

Use `generateStaticParams` to prerender all locale variants at build time:

```tsx
// app/[lang]/products/page.tsx
export async function generateStaticParams() {
  const locales = ['en-US', 'fr-FR', 'de-DE', 'ja-JP']
  const products = await getProducts()

  return locales.flatMap((lang) =>
    products.map((product) => ({
      lang,
      slug: product.slug,
    }))
  )
}

export default async function ProductsPage({
  params,
}: {
  params: Promise<{ lang: string; slug: string }>
}) {
  const { lang, slug } = await params
  const dict = await getDictionary(lang)
  const product = await getProduct(slug)

  return (
    <div>
      <h1>{product.name}</h1>
      <p>{dict.products.price}: ${product.price}</p>
    </div>
  )
}
```

## Domain Routing (Alternative)

For domain-based routing (e.g., `myapp.fr`, `myapp.de`):

```ts
// next.config.ts
const nextConfig = {
  domains: [
    { domain: 'myapp.com', defaultLocale: 'en-US' },
    { domain: 'myapp.fr', defaultLocale: 'fr-FR' },
    { domain: 'myapp.de', defaultLocale: 'de-DE' },
    { domain: 'myapp.jp', defaultLocale: 'ja-JP', locales: ['ja-JP'] },
  ],
}
```

Proxy locale detection still applies for sub-path fallback:

```ts
// proxy.ts
export async function proxy(request: NextRequest) {
  // Domain routing handles primary locale detection
  // Proxy adds fallback for sub-paths
  const { pathname } = request.nextUrl
  // ...
}
```

## Language Switcher Component

```tsx
// app/[lang]/language-switcher.tsx
'use client'

import { usePathname, useRouter } from 'next/navigation'

const localeNames: Record<string, string> = {
  'en-US': 'English',
  'fr-FR': 'Français',
  'de-DE': 'Deutsch',
  'ja-JP': '日本語',
}

export function LanguageSwitcher({ currentLang }: { currentLang: string }) {
  const router = useRouter()
  const pathname = usePathname()

  function switchLocale(newLocale: string) {
    // Replace current locale in pathname with new locale
    const newPath = pathname.replace(`/${currentLang}`, `/${newLocale}`)
    router.push(newPath)
  }

  return (
    <select
      value={currentLang}
      onChange={(e) => switchLocale(e.target.value)}
      aria-label="Select language"
    >
      {Object.entries(localeNames).map(([code, name]) => (
        <option key={code} value={code}>
          {name}
        </option>
      ))}
    </select>
  )
}
```

```tsx
// app/[lang]/layout.tsx
export default async function LocaleLayout({ children, params }: Props) {
  const { lang } = await params
  const dict = await getDictionary(lang)
  const isRTL = ['ar', 'he', 'fa'].includes(lang.split('-')[0])

  return (
    <html lang={lang} dir={isRTL ? 'rtl' : 'ltr'}>
      <body>
        {/* RTL-aware styling: flex-direction, text-align, padding, etc. */}
        {children}
      </body>
    </html>
  )
}
```

## Gotchas

1. **Locale detection via Accept-Language** — requires Proxy running on all routes; be careful with `matcher` exclusions
2. **Static generation requires `generateStaticParams`** — for all locale variants of dynamic routes
3. **Translation files are server-side only** — they never ship to the client; only the rendered HTML is sent
4. **`params.lang` is a string** — always validate with `hasLocale()` before using
5. **URLs must include locale prefix** — every localized page should have `/${lang}/` in its path
6. **Static imports of dictionaries** — use `import()` (dynamic) not static import to avoid bundling all locales in client
7. **Default locale handling** — decide whether to redirect `/` to `/${defaultLocale}` or serve default locale without prefix
